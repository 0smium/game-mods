cmake_minimum_required(VERSION 3.24)
project(GameMods LANGUAGES C CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Runtime requires MSVC x64 (Visual Studio 2022 C++ tools).")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(SMTVFIX_SOURCE "" CACHE PATH "Optional existing pinned SMTVFix 1.0.0 checkout with submodules")
if(NOT SMTVFIX_SOURCE)
  include(FetchContent)
  FetchContent_Declare(smtvfix
    GIT_REPOSITORY https://codeberg.org/Lyall/SMTVFix.git
    GIT_TAG c9eff87080b5fd87199c0835670dde2aba293184
    GIT_SUBMODULES external/safetyhook external/zydis
    GIT_SUBMODULES_RECURSE TRUE)
  FetchContent_GetProperties(smtvfix)
  if(NOT smtvfix_POPULATED)
    FetchContent_Populate(smtvfix)
  endif()
  set(SMTVFIX_SOURCE "${smtvfix_SOURCE_DIR}")
endif()
set(ZYDIS_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ZYDIS_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(ZYDIS_BUILD_DOXYGEN OFF CACHE BOOL "" FORCE)
set(ZYDIS_BUILD_SHARED_LIB OFF CACHE BOOL "" FORCE)
add_subdirectory("${SMTVFIX_SOURCE}/external/zydis" "zydis" EXCLUDE_FROM_ALL)
set(SAFETYHOOK_ROOT "${SMTVFIX_SOURCE}/external/safetyhook")
add_library(camera_safetyhook STATIC
  "${SAFETYHOOK_ROOT}/src/allocator.cpp" "${SAFETYHOOK_ROOT}/src/easy.cpp"
  "${SAFETYHOOK_ROOT}/src/inline_hook.cpp" "${SAFETYHOOK_ROOT}/src/mid_hook.cpp"
  "${SAFETYHOOK_ROOT}/src/os.windows.cpp" "${SAFETYHOOK_ROOT}/src/utility.cpp"
  "${SAFETYHOOK_ROOT}/src/vmt_hook.cpp")
target_include_directories(camera_safetyhook PUBLIC "${SAFETYHOOK_ROOT}/include")
target_link_libraries(camera_safetyhook PUBLIC Zydis)
add_library(SMTVVCameraRuntime SHARED mods/smtvv/common/runtime.cpp)
target_include_directories(SMTVVCameraRuntime PRIVATE "${SMTVFIX_SOURCE}/src")
target_compile_definitions(SMTVVCameraRuntime PRIVATE NOMINMAX)
target_compile_options(SMTVVCameraRuntime PRIVATE /W4 /EHsc /utf-8 /bigobj)
target_link_libraries(SMTVVCameraRuntime PRIVATE camera_safetyhook user32)
foreach(feature IN ITEMS FirstPerson GardenFreecam)
  if(feature STREQUAL "FirstPerson")
    set(feature_dir "first-person")
  else()
    set(feature_dir "garden-freecam-visibility")
  endif()
  add_library(SMTVV${feature} SHARED mods/smtvv/${feature_dir}/entry.cpp)
  target_link_libraries(SMTVV${feature} PRIVATE SMTVVCameraRuntime)
  set_target_properties(SMTVV${feature} PROPERTIES PREFIX "" SUFFIX ".asi")
endforeach()
